  <script>
    let myChart = null;
    let isHidden = false;
    let rawDataStore = null;
    let deferredPrompt = null;

    const baseUrl = "<?= scriptUrl ?>";

    const listKategoriPengeluaran = [
      "Belanja Dapur & Sembako", "Tagihan (Listrik/Air/Wifi)", "Kebutuhan Anak", 
      "Transportasi", "Kesehatan", "Cicilan & Hutang", "Jajan & Hiburan", 
      "Sosial & Agama", "Tabungan & Investasi", "Pengeluaran Lain-lain"
    ];

    const listKategoriPemasukan = [
      "Gaji Utama", "Bonus & Sampingan", "Hasil Investasi", "Pemasukan Lain-lain"
    ];

    // Format Rupiah
    function formatRupiah(num) {
      if (isHidden) return "••••••••";
      return new Intl.NumberFormat('id-ID', { style: 'currency', currency: 'IDR', maximumFractionDigits: 0 }).format(num || 0);
    }

    // Toggle Privacy (Intip / Sembunyikan Angka)
    function togglePrivacy() {
      isHidden = !isHidden;
      const icon = document.getElementById('iconPrivacy');
      if (isHidden) {
        icon.className = "fa-solid fa-eye-slash text-amber-400";
      } else {
        icon.className = "fa-solid fa-eye text-slate-300";
      }
      if (rawDataStore) renderDashboard(rawDataStore);
    }

    // Inisialisasi Aplikasi & PWA
    window.onload = function() {
      setupPWA();

      const hariIni = new Date();
      const bulanStr = ("0" + (hariIni.getMonth() + 1)).slice(-2);
      const thnStr = hariIni.getFullYear().toString();

      document.getElementById('filterBulan').value = bulanStr;
      document.getElementById('filterTahun').value = thnStr;
      document.getElementById('txtTanggal').valueAsDate = hariIni;

      updateKategoriOptions();
      muatDataRingkasan();
    };

    // Registrasi PWA Service Worker & Manifest
    function setupPWA() {
      if (baseUrl) {
        document.getElementById('pwaManifestLink').href = baseUrl + "?pwa=manifest";

        if ('serviceWorker' in navigator) {
          navigator.serviceWorker.register(baseUrl + "?pwa=sw")
            .then(reg => console.log('Service Worker Registered Dynamic'))
            .catch(err => console.log('SW Reg Failed:', err));
        }
      }

      window.addEventListener('beforeinstallprompt', (e) => {
        e.preventDefault();
        deferredPrompt = e;
        document.getElementById('btnInstallPwa').classList.remove('hidden');
      });
    }

    function installPWA() {
      if (deferredPrompt) {
        deferredPrompt.prompt();
        deferredPrompt.userChoice.then((choiceResult) => {
          if (choiceResult.outcome === 'accepted') {
            document.getElementById('btnInstallPwa').classList.add('hidden');
          }
          deferredPrompt = null;
        });
      }
    }

    function updateKategoriOptions() {
      const tipe = document.getElementById('selTipe').value;
      const selKat = document.getElementById('selKategori');
      selKat.innerHTML = '';

      const list = (tipe === 'Pengeluaran') ? listKategoriPengeluaran : listKategoriPemasukan;
      list.forEach(kat => {
        const opt = document.createElement('option');
        opt.value = kat;
        opt.textContent = kat;
        selKat.appendChild(opt);
      });
    }

    function muatDataRingkasan() {
      const bulan = document.getElementById('filterBulan').value;
      const tahun = document.getElementById('filterTahun').value;

      google.script.run
        .withSuccessHandler(res => {
          rawDataStore = res;
          renderDashboard(res);
        })
        .withFailureHandler(err => alert("Gagal mengambil data: " + err.message))
        .getRingkasan(bulan, tahun);
    }

    function renderDashboard(res) {
      document.getElementById('txtSaldoTotal').textContent = formatRupiah(res.saldo);
      document.getElementById('txtSaldoBulanLalu').textContent = formatRupiah(res.saldoBulanLalu);
      document.getElementById('txtPemasukan').textContent = formatRupiah(res.pemasukan);
      document.getElementById('txtPengeluaran').textContent = formatRupiah(res.pengeluaran);

      // Render Ember 50/30/20
      const ideal = res.alokasiRekomendasi;
      
      const pctPokok = ideal.pokokIdeal > 0 ? Math.min(100, (res.pakaiPokok / ideal.pokokIdeal) * 100) : 0;
      document.getElementById('barPokok').style.width = pctPokok + '%';
      document.getElementById('txtMetaPokok').textContent = `${formatRupiah(res.pakaiPokok)} / ${formatRupiah(ideal.pokokIdeal)}`;

      const pctGaya = ideal.gayaHidupIdeal > 0 ? Math.min(100, (res.pakaiGayaHidup / ideal.gayaHidupIdeal) * 100) : 0;
      document.getElementById('barGaya').style.width = pctGaya + '%';
      document.getElementById('txtMetaGaya').textContent = `${formatRupiah(res.pakaiGayaHidup)} / ${formatRupiah(ideal.gayaHidupIdeal)}`;

      const pctMasaDepan = ideal.masaDepanIdeal > 0 ? Math.min(100, (res.pakaiMasaDepan / ideal.masaDepanIdeal) * 100) : 0;
      document.getElementById('barMasaDepan').style.width = pctMasaDepan + '%';
      document.getElementById('txtMetaMasaDepan').textContent = `${formatRupiah(res.pakaiMasaDepan)} / ${formatRupiah(ideal.masaDepanIdeal)}`;

      // Render Tabel Transaksi
      const tbody = document.getElementById('tabelRiwayat');
      tbody.innerHTML = '';

      if (!res.riwayat || res.riwayat.length === 0) {
        tbody.innerHTML = '<tr><td colspan="6" class="p-4 text-center text-slate-500 text-xs">Belum ada transaksi di periode ini.</td></tr>';
      } else {
        res.riwayat.forEach(item => {
          const isExp = item.tipe === 'Pengeluaran';
          const tr = document.createElement('tr');
          tr.className = 'hover:bg-slate-800/40';
          tr.innerHTML = `
            <td class="p-3 text-slate-400 text-xs">${item.tanggal}</td>
            <td class="p-3"><span class="px-2 py-0.5 text-[10px] font-semibold rounded-md ${isExp ? 'bg-rose-500/10 text-rose-400' : 'bg-emerald-500/10 text-emerald-400'}">${isExp ? 'EXPENSE' : 'INCOME'}</span></td>
            <td class="p-3 font-medium">${item.kategori}</td>
            <td class="p-3 text-slate-400 text-xs">${item.keterangan || '-'}</td>
            <td class="p-3 text-right font-semibold ${isExp ? 'text-rose-400' : 'text-emerald-400'}">${isExp ? '-' : '+'}${formatRupiah(item.nominal)}</td>
            <td class="p-3 text-center">
              <button onclick="hapusTransaksi(${item.rowIndex})" class="text-slate-500 hover:text-rose-400 transition"><i class="fa-solid fa-trash"></i></button>
            </td>
          `;
          tbody.appendChild(tr);
        });
      }

      renderChart(res.kategoriData);
    }

    function renderChart(kategoriData) {
      const ctx = document.getElementById('kategoriChart').getContext('2d');
      const labels = Object.keys(kategoriData || {});
      const values = Object.values(kategoriData || {});

      if (myChart) myChart.destroy();

      if (labels.length === 0) {
        ctx.clearRect(0, 0, 300, 150);
        return;
      }

      myChart = new Chart(ctx, {
        type: 'doughnut',
        data: {
          labels: labels,
          datasets: [{
            data: values,
            backgroundColor: ['#6366f1', '#ec4899', '#f59e0b', '#10b981', '#06b6d4', '#8b5cf6', '#f43f5e']
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: { legend: { position: 'right', labels: { color: '#94a3b8', font: { size: 10 } } } }
        }
      });
    }

    function handleSimpan(e) {
      e.preventDefault();
      const btn = document.getElementById('btnSimpan');
      btn.disabled = true;
      btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin mr-2"></i>Menyimpan...';

      const payload = {
        tanggal: document.getElementById('txtTanggal').value,
        tipe: document.getElementById('selTipe').value,
        kategori: document.getElementById('selKategori').value,
        nominal: document.getElementById('txtNominal').value,
        keterangan: document.getElementById('txtKeterangan').value
      };

      google.script.run
        .withSuccessHandler(msg => {
          document.getElementById('txtNominal').value = '';
          document.getElementById('txtKeterangan').value = '';
          btn.disabled = false;
          btn.innerHTML = 'Simpan Transaksi';
          muatDataRingkasan();
        })
        .withFailureHandler(err => {
          alert("Gagal menyimpan: " + err.message);
          btn.disabled = false;
          btn.innerHTML = 'Simpan Transaksi';
        })
        .simpanData(payload);
    }

    function hapusTransaksi(rowIndex) {
      if (confirm("Apakah Anda yakin ingin menghapus transaksi ini?")) {
        google.script.run
          .withSuccessHandler(() => muatDataRingkasan())
          .hapusData(rowIndex);
      }
    }

    function bukaModalClearAll() {
      if (confirm("🚨 WARNING: Seluruh data transaksi di Google Sheet akan DIHAPUS PERMANEN. Lanjutkan?")) {
        google.script.run
          .withSuccessHandler(msg => {
            alert(msg);
            muatDataRingkasan();
          })
          .clearAllData();
      }
    }
  </script>
